267# This function calculates the SNR of a signal knowing what the signal should

269# -----
270# ideal - the signal with no noise
271# actual - the signal with noise
272def getSNR (ideal, actual):
273 # Verify signals have same shape
274 if (np.shape(ideal)[0] != np.shape(actual)[0]):
275 print("Cannot go forward: Ideal and Actual signals are not the same length.")
276 print("Ideal: %d samples. Actual: %d samples." % (np.shape(ideal)[0], np.shape(actual)[0]))

278
279 # Find noise SNR and signal SNR
280 noise = np.subtract(actual, ideal)
281 n_samps = np.shape(ideal)[0]
282 avg_noise_pwr = np.real(np.sum(np.multiply(noise, np.conj(noise)))) / n_samps
283 avg_signal_pwr = np.real(np.sum(np.multiply(ideal, np.conj(ideal)))) / n_samps
284
285 snr = avg_signal_pwr / avg_noise_pwr
286 return snr
