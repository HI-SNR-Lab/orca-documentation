206void configureRate(boost::asio::serial_port& serial, int hz) {
207 uint16_t measRate = 1000 / hz;
208 std::vector<uint8_t> msg = {
209 0xB5, 0x62, 0x06, 0x08, 0x06, 0x00,
210 static_cast<uint8_t>(measRate & 0xFF),
211 static_cast<uint8_t>((measRate >> 8) & 0xFF),
212 0x01, 0x00, 0x01, 0x00, // navRate = 1, timeRef = UTC
213 0x00, 0x00 // Checksum placeholders
214 };
215
216 // Calculate checksum
217 uint8_t ckA = 0, ckB = 0;
218 for (size_t i = 2; i < 12; ++i) {
219 ckA += msg[i];
220 ckB += ckA;
221 }
222 msg[12] = ckA;
223 msg[13] = ckB;
224
225 sendUBX(serial, msg);
