6def add_distance_to_xarray(data_array, slow_time_values, gps_times, gps_distances):
7 """
8 Add distance-along-track coordinate to xarray by interpolating GPS data.
9
10 Parameters
11 ----------
12 data_array : xr.DataArray
13 2D xarray with dimensions (sample_idx, ...) and slow_time coordinate
14 slow_time_values : array-like
15 The slow_time values corresponding to each sample_idx
16 gps_times : array-like
17 Time values from GPS data
18 gps_distances : array-like
19 Distance along track values from GPS data
20
21 Returns
22 -------
23 xr.DataArray
24 Data array with added 'distance_along_track' coordinate
25 """
26 # Create interpolation function for distance along track
27 # Use bounds_error=False to extrapolate for times outside GPS range
28 distance_interp = interp1d(
29 gps_times,
30 gps_distances,
31 kind='linear',
32 bounds_error=False,
33 fill_value='extrapolate'
34 )
35
36 # Interpolate distances for each slow_time
37 distances = distance_interp(slow_time_values)
38
39 # Add as a coordinate to the data array
40 data_with_distance = data_array.assign_coords(
41 distance_along_track=('pulse_idx', distances)
42 )
43
44 return data_with_distance

47def group_and_average_by_distance(data_array, bin_size=5.0, distance_coord='distance_along_track'):
48 """
49 Group xarray columns by distance along track and compute mean.
50
51 Parameters
52 ----------
53 data_array : xr.DataArray
54 Data array with distance_along_track coordinate
55 bin_size : float
56 Size of distance bins in meters (default: 5.0)
57 distance_coord : str
58 Name of the distance coordinate (default: 'distance_along_track')
59
60 Returns
61 -------
62 xr.DataArray
63 Averaged data grouped by distance bins
64 """
65 # Get the distance values
66 distances = data_array[distance_coord].values
67
68 # Create bins
69 min_dist = np.floor(distances.min())
70 max_dist = np.ceil(distances.max())
71 bins = np.arange(min_dist, max_dist + bin_size, bin_size)
72
73 # Assign each sample to a bin (using bin centers as labels)
74 bin_indices = np.digitize(distances, bins) - 1
75 bin_centers = bins[:-1] + bin_size / 2
76
77 # Clip to valid range
78 bin_indices = np.clip(bin_indices, 0, len(bin_centers) - 1)
79
80 # Add bin assignment as a coordinate
81 data_with_bins = data_array.assign_coords(
82 distance_bin=('pulse_idx', bin_centers[bin_indices])
83 )
84
85 # Group by distance bin and compute mean
86 grouped = data_with_bins.groupby('distance_bin').mean(dim='pulse_idx')
87
88 return grouped
