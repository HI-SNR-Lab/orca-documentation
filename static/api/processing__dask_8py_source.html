13def process_stdout_log(log):
14 """
15 Load timestamp and ERROR_CODE_LATE_COMMAND information from UHD radar code stdout.
16 Returns the starting timestamp the number of pulses attempted and a dictionary of errors, where the keys are
17 chirp indices and the values are error codes.
18 """
19 errors = {}
20 start_timestamp = None
21 num_pulses_attempted = None
22
23 for idx, line in enumerate(log.splitlines()):
24 if "Receiver error:" in line:
25 error_code = re.search(
26 r"(?:Receiver error: )([\w_]+)", line).groups()[0]
27 old_style_regex_search = re.search(
28 r"(?:Scheduling chirp )([\d]+)", log[idx-1])
29 if old_style_regex_search is not None:
30 chirp_idx = int(
31 re.search(r"(?:Scheduling chirp )([\d]+)", log[idx-1]).groups()[0])
32 else:
33 chirp_idx = int(
34 re.search(r"(?:Chirp )([\d]+)", line).groups()[0])
35 errors[chirp_idx] = error_code
36 if error_code != "ERROR_CODE_LATE_COMMAND":
37 print(
38 f"WARNING: Uncommon error in the log: {error_code} (on chirp {chirp_idx})")
39 print(f"Full message: {line}")
40 if ("[START]" in line) or ("Scheduling chirp 0 RX" in line):
41 start_timestamp = float(
42 re.search(r"(?:\[)([\d]+\.[\d]+)", line).groups()[0])
43 if ("Total pulses attempted" in line):
44 num_pulses_attempted = int(re.search(r": ([\d]+)", line).groups()[0])
45
46 return start_timestamp, errors, num_pulses_attempted

49def save_radar_data_to_zarr(prefix, skip_if_cached=True, zarr_base_location=None, expected_base_name_regex=r'\d{8}_\d{6}', log_required=True, dryrun=False):
50 """
51 Load raw radar data from a given prefix, and save it to a zarr file.
52
53 `prefix` is the path to the raw data, without the _rx_samps.bin/_config.yaml/_uhd_stdout.log suffixes.
54 (`log_required` can be set to False if no log file is available)
55
56 As a safety precaution, this function will check that the prefix basename matches the `expected_base_name_regex` expression.
57 If using the python code to run the radar system, the prefixes will be of the form YYYYMMDD_HHMMSS,
58 which is what the default regex is looking for. `expected_base_name_regex` can be set to None to skip this check.
59
60 The location for the zarr file is the same directory containing the prefix, unless you provide
61 an alternate `zarr_base_location` argument.
62
63 By default, this will first look for an existing zarr file that matches the expected filename.
64 If you want to force reprocessing, set `skip_if_cached` to False.
65
66 Setting `dryrun` to True will cause this function to return the path to the zarr file
67 that it would have created without actually writing anything to disk.
68
69 Returns the path to the zarr file only. You are responsible for re-loading the data from the zarr file.
70 """
71
72 #
73 # Validation and file name generation
74 #
75 basename = os.path.basename(prefix)
76 if expected_base_name_regex: # Optional sanity check that basename makes sense -- set expected_base_name_regex to None to skip
77 if not re.match(expected_base_name_regex, basename):
78 raise ValueError(
79 f"Prefix basename {basename} does not match expected regex {expected_base_name_regex}")
80
81 # Generate expected zarr output location
82 if zarr_base_location is None:
83 zarr_path = prefix + ".zarr"
84 else:
85 zarr_path = os.path.join(zarr_base_location, basename + ".zarr")
86
87 # Check if zarr file already exists, if so just return the path
88 if skip_if_cached and os.path.exists(zarr_path):
89 return zarr_path
90
91 # Build filenames from prefix
92 rx_samps_file = prefix + "_rx_samps.bin"
93 log_file = prefix + "_uhd_stdout.log"
94 gpsd_file = prefix + "_gpspipe_stdout.log"
95
96 #
97 # Data loading
98 #
99 # Load configuration YAML
100 config = old_processing.load_config(prefix)
101
102 cpu_format = config['DEVICE'].get('cpu_format', 'fc32')
103 if cpu_format == 'fc32':
104 output_dtype = np.float32
105 scale_factor = 1.0
106 elif cpu_format == 'sc16':
107 output_dtype = np.int16
108 scale_factor = np.iinfo(output_dtype).max
109 elif cpu_format == 'sc8':
110 output_dtype = np.int8
111 scale_factor = np.iinfo(output_dtype).max
112 else:
113 raise Exception(f"Unrecognized cpu_format '{cpu_format}'. Must be one of 'fc32', 'sc16', or 'sc8'.")
114
115 # Load raw RX samples
116 rx_len_samples = int(config['CHIRP']['rx_duration']
117 * config['GENERATE']['sample_rate'])
118 rx_sig = da.from_array(
119 np.memmap(rx_samps_file, dtype=output_dtype, mode='r', order='C'), chunks=rx_len_samples*2*100)
120 rx_sig = (rx_sig[::2] + (1j * rx_sig[1::2])).astype(np.complex64) / scale_factor
121 n_rxs = rx_sig.size // rx_len_samples
122 radar_data = da.transpose(da.reshape(
123 rx_sig, (n_rxs, rx_len_samples), merge_chunks=True))
124
125 # Create time axes
126 slow_time = np.linspace(0, config['CHIRP']['pulse_rep_int']
127 * config['CHIRP'].get('num_presums', 1)*n_rxs, radar_data.shape[1])
128 fast_time = np.linspace(
129 0, config['CHIRP']['rx_duration'], radar_data.shape[0])
130
131 # Load raw data from log
132 log = None
133 if os.path.exists(log_file):
134 with open(log_file, 'r') as log_f:
135 log = log_f.read()
136 else:
137 if log_required:
138 raise FileNotFoundError(
139 f"Log file not found: {log_file}. If a log file is not required, set log_required=False")
140
141 gpsdlog = None
142 if os.path.exists(gpsd_file):
143 with open(gpsd_file, 'r') as gpsd_f:
144 gpsdlog = gpsd_f.read()
145 else:
146 print("No GPS log file found!!")
147 gpsdlog = None
148 #if log_required:
149 # raise FileNotFoundError(
150 # f"Log file not found: {log_file}. If a log file is not required, set log_required=False")
151
152
153 # Save radar_data, slow_time, and fs to an xarray datarray
154 data = xr.Dataset(
155 data_vars={
156 "radar_data": (["sample_idx", "pulse_idx"], radar_data, {"description": "complex radar data"}),
157 },
158 coords={
159 "sample_idx": ("sample_idx", np.arange(radar_data.shape[0]), {"description": "Index of this sample in the chirp"}),
160 "fast_time": ("sample_idx", fast_time, {"description": "time relative to start of this recording interval in seconds"}),
161 "pulse_idx": ("pulse_idx", np.arange(radar_data.shape[1]), {"description": "Index of this chirp in the sequence"}),
162 "slow_time": ("pulse_idx", slow_time, {"description": "time in seconds"}),
163 },
164 attrs={
165 "config": config,
166 "stdout_log": log,
167 "gpspipe_log": gpsdlog,
168 "prefix": prefix,
169 "basename": basename
170 }
171 )
172
173 # TODO: Due to the currently hard-coded increase in pulse repetition interval after an error,
174 # the slow time may not be correct.
175
176 if not dryrun:
177 with dask.config.set(scheduler='single-threaded'):
178 data.to_zarr(zarr_path, mode="w")
179 else:
180 print("This is a dry run: not saving data to disk")
181 print(data)
182
183 return zarr_path

185def check_if_error_data_exists(data, errors=None, num_pulses_attempted=-1):
186 """
187
188 Attempts to automatically correct for an unintended behavior of some versions of the code.
189
190 Basically, there are two existant behaviors for the radar code when an ERROR_CODE_LATE_COMMAND
191 occurs. In one type, zeros or junk data of the lengths of a normal receive window are recorded.
192 In the other type, nothing is writen to the output file when an error occurs.
193
194 If a specific number of pulses was requested in the config file, we can automatically determine
195 which behavior occurred.
196
197 The intent of this function is to be used internally in error management functions (fill_errors
198 and remove_errors) to automatically prevent calling them in a way that doens't make sense.
199
200 Input variables:
201 data Xarray dataset, some format as everything else
202 errors Optional, if a dictionary of errors is already available from process_stdout_log,
203 you can provide it here to avoid calling that function twice.
204
205 There are four possible return values, returned as strings (sorry...)
206
207 "unknown" -- No specific number of pulses requsted, so there is not trivial way of determining
208 which case we're in. (Note that there are cases where we can still figure it out from the
209 log. This can be a future project. TODO)
210 "error_data_included" -- The number of recorded pulses exactly matches what was requested.
211 Note: This includes all recordings with 0 errors. In all intended use cases of this function,
212 it shouldn't matter what this function returns if there are zero errors.
213 "error_data_not_included" -- The number of recorded pulses exaclty matches what was requested minus
214 the number of errors.
215 "unexpected_data_length" -- The number of recorded pulses matches neither of the above criteria.
216 This could indicate that the recording was stopped early (by someone pressing Ctrl-C, for example)
217 or it could indicate a more serious problem. This return value should always be investigated.
218 """
219
220 pulses_requested = data.attrs['config']['CHIRP']['num_pulses']
221 pulses_in_data = len(data.pulse_idx) * data.attrs['config']['CHIRP']['num_presums']
222
223 if not errors:
224 _, errors, num_pulses_attempted = process_stdout_log(data.attrs["stdout_log"])
225
226 n_errors = len(errors)
227
228 if pulses_requested < 0:
229 return "unknown" # Number of attempted pulses unknown, so cannot trivially determine if error data is included
230 elif pulses_in_data == pulses_requested == num_pulses_attempted:
231 return "error_data_included"
232 elif pulses_in_data == (pulses_requested - n_errors) or pulses_requested == pulses_in_data == (num_pulses_attempted - n_errors):
233 return "error_data_not_included"
234 else:
235 return "unexpected_data_length"

237def fill_errors(data, error_fill_value=np.nan, allowed_file_error_types=[], force_file_error_type=None):
238 """
239 Replace all values from chirps with a reported error with the specified error_fill_value
240 """
241
242 _, errors, num_pulses_attempted = process_stdout_log(data.attrs["stdout_log"])
243 file_error_type = check_if_error_data_exists(data, errors, num_pulses_attempted)
244
245 if force_file_error_type != None: # force the file error type if we want control over error handling behavior
246 file_error_type = force_file_error_type
247
248 if file_error_type != "error_data_included":
249 if (file_error_type in allowed_file_error_types):
250 print(f"[WARNING] Unexpected file error type of {file_error_type} but " +
251 "explicitly allowed by allowed_file_error_types so proceeding without additional checks. This may have unexpected behaviors!")
252 else:
253 print(f"[WARNING] File error type is {file_error_type} so there's nothing for this function to do. Returning a copy of your unmodified dataset.")
254 return data.copy()
255
256 result = data.copy()
257 error_idxs = np.array(list(errors.keys()))
258
259 # only fill errors if they exist, otherwise avoid throwing an index error
260 if error_idxs.size != 0:
261 result["radar_data"][{"pulse_idx": error_idxs}] = error_fill_value
262
263 return result

313def pulse_compress(data: xr.Dataset, chirp, fs: float, zero_sample_idx: int=0, signal_speed: float=None):
314 """
315 Apply pulse compression using samples from `chirp` to each pulse from `data`.
316 Zero travel time is assumed to be at `zero_sample_idx` in the chirp.
317 If a `signal_speed` is provided, this is used to create a secondary coordinate
318 `reflection_distance` which is the distance from the radar to the reflection point
319 assuming a constant signal speed.
320 """
321
322 output_len = len(data["sample_idx"])-len(chirp)+1
323 travel_time = np.linspace(0, output_len/fs, output_len)
324 travel_time = travel_time - travel_time[zero_sample_idx]
325
326 coords = {"travel_time": travel_time}
327 if signal_speed is not None:
328 coords['reflection_distance'] = ("travel_time", travel_time * (signal_speed/2))
329
330 # This code is kind of a nightmare, but it should be a fairly efficient way
331 # to do this.
332 # This function call applies the lambda function (first argument) to each
333 # pulse in the data.
334 # If you want to understand all the other parameters, I recommend starting
335 # with these pages:
336 # https://docs.xarray.dev/en/stable/examples/apply_ufunc_vectorize_1d.html
337 # https://docs.xarray.dev/en/stable/generated/xarray.apply_ufunc.html
338
339 compressed = xr.apply_ufunc(
340 lambda x: scipy.signal.correlate(
341 x, chirp, mode='valid') / np.sum(np.abs(chirp)**2),
342 data,
343 input_core_dims=[['sample_idx']], # The dimension operated over -- aka "don't vectorize over this"
344 output_core_dims=[["travel_time"]], # The output dimensions of the lambda function itself
345 exclude_dims=set(("sample_idx",)), # Dimensions to not vectorize over
346 vectorize=True, # Vectorize other dimensions using a call to np.vectorize
347 dask="parallelized", # Allow dask to chunk and parallelize the computation
348 output_dtypes=[data["radar_data"].dtype], # Needed for dask: explicitly provide the output dtype
349 dask_gufunc_kwargs={"output_sizes": {'travel_time': output_len}} # Also needed for dask:
350 # explicitly provide the output size of the lambda function. See
351 # https://docs.dask.org/en/stable/generated/dask.array.gufunc.apply_gufunc.html
352 ).assign_coords(coords) # And finally add coordinate(s) corresponding to the new "travel_time" dimension
353
354 # Save the input parameters for future reference
355 compressed.attrs["pulse_compress"]={
356 "fs": fs, "zero_sample_idx": zero_sample_idx, "signal_speed": signal_speed}
357
358 return compressed

361def invert_phase_dithering(data, phase_codes_filename, override_errors=False):
362
363 if not override_errors:
364 if "phase_dithering_inversion" in data.attrs:
365 raise Exception("It looks like phase dithering inversion has already been run on this dataset.")
366 if not data.attrs['config']["CHIRP"].get("phase_dithering", False):
367 raise Exception("phase_dithering is not set in the config file. Are you sure you want to invert this file?")
368
369 phases = np.fromfile(phase_codes_filename, dtype=np.float32, count=len(data.pulse_idx))
370 xr_phases = xr.DataArray(phases, dims=('pulse_idx',))
371
372 demodulated = data.copy()
373
374 demodulated["radar_data"] *= np.exp(-1j * xr_phases)
375
376 demodulated.attrs["phase_dithering_inversion"] = {'phases': phases}
377
378 return demodulated
