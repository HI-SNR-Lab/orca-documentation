141/*** @brief Checks for 10 MHz reference lock
142*
143* This function checks if the USRP device has locked to the 10 MHz reference
144* signal. Retrieves the list of sensor names from the USRP device and checks
145* for the presence of the "ref_locked" sensor. If it exists, waits for
146* the reference lock to be established, printing a dot every second until
147* the lock is achieved or a timeout occurs. If the lock is successful, it
148* prints "LOCKED"; otherwise, it prints "FAILED" and exits the program.
149*
150*/
151void Sdr::check10MhzLock(){
152 vector<string> sensor_names = usrp->get_mboard_sensor_names(0);
153 if (find(sensor_names.begin(), sensor_names.end(), "ref_locked")
154 != sensor_names.end()) {
155 cout << "Waiting for reference lock..." << flush;
156 bool ref_locked = false;
157 for (int i = 0; i < 30 and not ref_locked; i++) {
158 ref_locked = usrp->get_mboard_sensor("ref_locked", 0).to_bool();
159 if (not ref_locked) {
160 cout << "." << flush;
161 this_thread::sleep_for(chrono::seconds(1));
162 }
163 }
164 if (ref_locked) {
165 cout << "LOCKED" << endl;
166 } else {
167 cout << "FAILED" << endl;
168 cout << "Failed to lock to GPSDO 10 MHz Reference. Exiting."
169 << endl;
170 exit(EXIT_FAILURE);
171 }
172 } else {
173 cout << boost::format(
174 "ref_locked sensor not present on this board.\n");
175 }
176}

178/*** @brief Locks GPS and sets the USRP time
179 *
180 * Waits for the GPS to lock, retrieves the GPS time, and sets the
181 * USRP time to the GPS time. If the GPS is locked, sets the USRP time,
182 * checks if the USRP time matches the GPS time and prints results. If
183 * GPS not locked, prints a warning message indicating that the time
184 * isn't accurate.
185 *
186*/
187void Sdr::gpsLock(){
188 //wait for GPS lock
189 bool gps_locked = usrp->get_mboard_sensor("gps_locked", 0).to_bool();
190 size_t num_gps_locked = 0;
191 for (int i = 0; i < 30 and not gps_locked; i++) {
192 gps_locked = usrp->get_mboard_sensor("gps_locked", 0).to_bool();
193 if (not gps_locked) {
194 cout << "." << flush;
195 this_thread::sleep_for(chrono::seconds(1));
196 }
197 }
198 if (gps_locked) {
199 num_gps_locked++;
200 cout << boost::format("GPS Locked\n");
201 } else {
202 cerr
203 << "WARNING: GPS not locked - time will not be accurate until locked"
204 << endl;
205 }
206}
207/*** @brief Checks the USRP time against GPS time
208 *
209 * Retrieves the GPS time from the USRP device, compares it with last PPS time
210 * from USRP device, and prints results. If the GPS time matches the last PPS time,
211 * synchronization is indicated. If it doesn't match, an error message is printed.
212 */
213void Sdr::checkAndSetTime(){
214 //set GPS time
215 time_spec_t gps_time = time_spec_t(int64_t(usrp->get_mboard_sensor("gps_time", 0).to_int()));
216 usrp->set_time_next_pps(gps_time + 1.0, 0);
217
218 // Wait for it to apply
219 // The wait is 2 seconds because N-Series has a known issue where
220 // the time at the last PPS does not properly update at the PPS edge
221 // when the time is actually set.
222 this_thread::sleep_for(chrono::seconds(2));
223 gps_time = time_spec_t(int64_t(usrp->get_mboard_sensor("gps_time", 0).to_int()));
224 time_spec_t time_last_pps = usrp->get_time_last_pps(0);
225 cout << "USRP time: "
226 << (boost::format("%0.9f") % time_last_pps.get_real_secs())
227 << endl;
228 cout << "GPSDO time: "
229 << (boost::format("%0.9f") % gps_time.get_real_secs()) << std::endl;
230 if (gps_time.get_real_secs() == time_last_pps.get_real_secs())
231 cout << endl
232 << "SUCCESS: USRP time synchronized to GPS time" << endl
233 << endl;
234 else
235 std::cerr << endl
236 << "ERROR: Failed to synchronize USRP time to GPS time"
237 << endl
238 << endl;
239}

243 * Splits the specified TX and RX channel strings into individual channel numbers,

247void Sdr::detectChannels(){
248 boost::split(tx_channel_strings, tx_channels, boost::is_any_of("\"',"));
249 for (size_t ch = 0; ch < tx_channel_strings.size(); ch++) {
250 size_t chan = stoi(tx_channel_strings[ch]);
251 if (chan >= usrp->get_tx_num_channels()) {
252 throw std::runtime_error("Invalid TX channel(s) specified.");
253 } else
254 tx_channel_nums.push_back(stoi(tx_channel_strings[ch]));
255 }
256 boost::split(rx_channel_strings, rx_channels, boost::is_any_of("\"',"));
257 for (size_t ch = 0; ch < rx_channel_strings.size(); ch++) {
258 size_t chan = stoi(rx_channel_strings[ch]);
259 if (chan >= usrp->get_rx_num_channels()) {
260 throw std::runtime_error("Invalid RX channel(s) specified.");
261 } else
262 rx_channel_nums.push_back(stoi(rx_channel_strings[ch]));
263 }
