10def generate_chirp(config):
11 """
12 Generate a chirp according to parameters in the config dictionary, typically
13 loaded from a config YAML file.
14
15 Returns a tuple (ts, chirp_complex), where ts is a numpy array of time
16 samples, and chirp_complex is a numpy array of complex floating point values
17 representing the chirp.
18
19 If you're looking for a floating point valued chirp to use in convolution,
20 this is probably the right function.
21
22 This function does not convert the complex numpy array to the cpu format
23 expected by the radar code. If you want to produce samples to feed the radar
24 code, look at `generate_from_yaml_filename` (later in this file) instead.
25 """
26 # Load parameters
27 gen_params = config["GENERATE"]
28 chirp_type = gen_params["chirp_type"]
29 sample_rate = gen_params["sample_rate"]
30 chirp_bandwidth = gen_params["chirp_bandwidth"]
31 offset = gen_params.get("lo_offset_sw", 0)
32 window = gen_params["window"]
33 chirp_length = gen_params["chirp_length"]
34 pulse_length = gen_params.get("pulse_length", chirp_length) # default to chirp_length is no pulse_length is specified
35
36 # Build chirp
37
38 end_freq = chirp_bandwidth / 2 # Chirp goes from -BW/2 to BW/2
39 start_freq = -1 * end_freq
40
41 start_freq += offset
42 end_freq += offset
43
44 ts = np.arange(0, chirp_length-(1/(2*sample_rate)), 1/(sample_rate))
45 ts_zp = np.arange(0, (pulse_length)-(1/(2*sample_rate)), 1/(sample_rate))
46
47 if chirp_type == 'linear':
48 ph = 2*np.pi*((start_freq)*ts + (end_freq - start_freq) * ts**2 / (2*chirp_length))
49 elif chirp_type == 'hyperbolic':
50 ph = 2*np.pi*(-1*start_freq*end_freq*chirp_length/(end_freq-start_freq))*np.log(1- (end_freq-start_freq)*ts/(end_freq*chirp_length))
51 else:
52 ph = 2*np.pi*(start_freq*ts + (end_freq - start_freq) * ts**2 / (2*chirp_length))
53 printf("[ERROR] Unrecognized chirp type '{chirp_type}'")
54 return None, None
55
56 chirp_complex = np.exp(1j*ph)
57
58 if window == "blackman":
59 chirp_complex = chirp_complex * np.blackman(chirp_complex.size)
60 elif window == "hamming":
61 chirp_complex = chirp_complex * np.hamming(chirp_complex.size)
62 elif window == "kaiser14":
63 chirp_complex = chirp_complex * np.kaiser(chirp_complex.size, 14.0)
64 elif window == "kaiser10":
65 chirp_complex = chirp_complex * np.kaiser(chirp_complex.size, 10.0)
66 elif window == "kaiser18":
67 chirp_complex = chirp_complex * np.kaiser(chirp_complex.size, 18.0)
68 elif window != "rectangular":
69 print("[ERROR] Unrecognized window function '{window}'")
70 return None, None
71
72 chirp_complex = np.pad(chirp_complex, (int(np.floor(ts_zp.size - ts.size)/2),), 'constant')
73
74 chirp_complex = chirp_complex
75
76 return ts_zp, chirp_complex

79def generate_from_yaml_filename(yaml_filename):
