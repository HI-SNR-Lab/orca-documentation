70def plot_matched(matched, sample_rate, tx=None, velocity_factor=2/3, loopback=True, coax_length=None, zero_sample_idx=159):
71 """Plot matched filter output with physical distance scaling.
72
73 Args:
74 matched: 2D array (samples x pulses) after matched filtering.
75 sample_rate: sampling rate in Hz.
76 tx: optional transmit chirp (1D array). If provided, its length is
77 used to compute the delay offset. If not, a fallback is used.
78 velocity_factor: fraction of light speed inside the medium (coax),
79 default 2/3 for typical coax.
80 loopback: True if the signal traverses the medium one-way (tx->rx).
81 For free-space reflections set False (two-way and divide by 2).
82 coax_length: optional expected coax length in meters; if provided,
83 a vertical line and annotation for expected return is drawn.
84 """
85 c0 = 299792458.0 # speed of light in vacuum (m/s)
86
87 range_profile = np.sum(matched, axis=1) # coherent sum across matched axis
88 magnitude = np.abs(range_profile)
89
90 # Use the provided chirp length if available, otherwise fall back to 1120
91 chirp_len = len(tx) if tx is not None else 1120
92 delay_offset = chirp_len - 1 + zero_sample_idx # correct for the delay caused by the matched filter, 159 comes from hardware latency and is proprotional to sampling rate (taken from Loopback test jupyter notebook)
93
94 n = np.arange(len(magnitude))
95 # time per sample converted from sample index difference
96 time = (n - delay_offset) / float(sample_rate)
97
98 v = velocity_factor * c0
99 if loopback:
100 ranges = time * v
101 else:
102 ranges = time * v / 2.0
103
104 v = 2/3 * 299792458
105 #Change this variable if your cable isn't 50 meters
106 cable_length = 50
107 expected_idx = int(cable_length / v * sample_rate + len(tx) - 1)
108 print("Expected sample index for", cable_length, "m:", expected_idx)
109
110 mask = ranges >= 0
111 magnitude = magnitude[mask]
112 ranges = ranges[mask]
113 magnitude = 20 * np.log10(magnitude)
114
115 plt.figure()
116 plt.plot(ranges, magnitude)
117 plt.title("Matched Filter")
118 plt.xlabel("Distance (meters)")
119 plt.ylabel("Normalized Amplitude")
120 plt.grid()
121
122 # If expected physical length is provided, mark it on the plot
123 if coax_length is not None:
124 expected_time = coax_length / v
125 expected_sample = expected_time * sample_rate + delay_offset
126 expected_range = coax_length
127 plt.axvline(expected_range, color='r', linestyle='--', label=f"expected {coax_length} m")

129 # Diagnostics: find the received peak and suggest a velocity_factor correction
